# 学校物品二维码管理系统

物品录入 → 借出 → 归还 → 报废 全生命周期管理

---

## 本地启动后端

### 1. 安装依赖

```bash
cd school-qrcode-system
pip install -r requirements.txt
```

### 2. 初始化数据库（首次运行）

```bash
cd backend
python init_db.py
```

默认创建以下测试账号：

| 用户名    | 密码      | 角色    |
|-----------|-----------|---------|
| admin     | admin123  | 管理员  |
| teacher1  | teacher1  | 教职工  |
| student1  | student1  | 学生    |

### 3. 启动后端

```bash
cd backend
python main.py
```

默认端口 **8000**，API 文档：http://localhost:8000/docs

### 4. 使用自定义端口

通过环境变量设置：

```bash
# macOS / Linux
API_PORT=9090 python main.py

# 或者 export
export API_PORT=9090
python main.py
```

### 5. 所有可配置环境变量

| 变量名         | 默认值                          | 说明                        |
|----------------|--------------------------------|-----------------------------|
| API_PORT       | 8000                           | 后端 API 端口               |
| DB_PATH        | data/school_qr.db              | SQLite 数据库文件路径        |
| SESSION_SECRET | dev-secret-change-in-prod      | Session 签名密钥             |
| SERVER_HOST    | localhost:8080                  | 二维码中的前端地址           |
| STATIC_DIR     | static                         | 静态文件目录（二维码/PDF）   |

### 6. 验证后端运行

```bash
# 健康检查
curl http://localhost:8000/

# 登录（获取 session cookie）
curl -c cookies.txt -X POST http://localhost:8000/api/auth/login \
  -H "Content-Type: application/json" \
  -d '{"username":"admin","password":"admin123"}'

# 查看物品列表（带 cookie）
curl -b cookies.txt http://localhost:8000/api/items

# 查看仪表盘统计
curl -b cookies.txt http://localhost:8000/api/dashboard/stats
```

```python
"""封装后端 API 调用 — 每次请求从 NiceGUI app.storage.user 读 session_id"""
import os
import httpx
from nicegui import app

API_BASE = os.getenv("API_BASE_URL", "http://localhost:8000/api")


def _get_session_id() -> str | None:
    """从 NiceGUI 用户存储读取 session_id"""
    try:
        return app.storage.user.get("session_id")
    except Exception:
        return None


def _cookies() -> dict:
    """构造 Cookie dict"""
    sid = _get_session_id()
    return {"session_id": sid} if sid else {}


def _post(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.post(path, cookies=_cookies(), **kwargs)
        return resp.json()


def _get(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.get(path, cookies=_cookies(), **kwargs)
        return resp.json()


def _put(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.put(path, cookies=_cookies(), **kwargs)
        return resp.json()


def _delete(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.delete(path, cookies=_cookies(), **kwargs)
        return resp.json()


class ApiClient:

    # ---------- 认证 ----------
    def login(self, username: str, password: str) -> dict:
        """登录 — 特殊处理：需要从 response 提取 session_id cookie"""
        with httpx.Client(base_url=API_BASE, timeout=30) as c:
            resp = c.post("/auth/login", json={"username": username, "password": password})
            data = resp.json()
            # 把后端返回的 session_id cookie 存到 NiceGUI user storage
            sid = resp.cookies.get("session_id")
            if sid:
                app.storage.user["session_id"] = sid
            return data

    def logout(self) -> dict:
        result = _post("/auth/logout")
        try:
            app.storage.user.pop("session_id", None)
        except Exception:
            pass
        return result

    def get_me(self) -> dict:
        return _get("/auth/me")

    # ---------- 物品 ----------
    def list_items(self, **params) -> dict:
        return _get("/items", params=params)

    def get_item(self, item_id: int) -> dict:
        return _get(f"/items/{item_id}")

    def create_item(self, data: dict) -> dict:
        return _post("/items", json=data)

    def update_item(self, item_id: int, data: dict) -> dict:
        return _put(f"/items/{item_id}", json=data)

    def delete_item(self, item_id: int) -> dict:
        return _delete(f"/items/{item_id}")

    def get_categories(self) -> dict:
        return _get("/items/categories")

    def get_qrcode_url(self, item_id: int) -> str:
        """返回二维码图片的完整 URL（含 session cookie 无法直接用，走代理）"""
        return f"{API_BASE.replace('/api', '')}/api/items/{item_id}/qrcode"

    # ---------- 借用 ----------
    def borrow_item(self, item_id: int, due_date: str, purpose: str = "") -> dict:
        return _post("/borrows", json={
            "item_id": item_id, "due_date": due_date, "purpose": purpose,
        })

    def return_item(self, borrow_id: int, return_note: str = "") -> dict:
        return _put(f"/borrows/{borrow_id}/return", json={"return_note": return_note})

    def list_borrows(self, **params) -> dict:
        return _get("/borrows", params=params)

    def my_borrows(self, **params) -> dict:
        return _get("/borrows/my", params=params)

    def list_overdue(self) -> dict:
        return _get("/borrows/overdue")

    # ---------- 报废 ----------
    def apply_scrap(self, item_id: int, reason: str) -> dict:
        return _post("/scraps", json={"item_id": item_id, "reason": reason})

    def approve_scrap(self, scrap_id: int, note: str = "") -> dict:
        return _put(f"/scraps/{scrap_id}/approve", json={"approve_note": note})

    def reject_scrap(self, scrap_id: int, note: str = "") -> dict:
        return _put(f"/scraps/{scrap_id}/reject", json={"approve_note": note})

    def list_scraps(self, **params) -> dict:
        return _get("/scraps", params=params)

    # ---------- 用户 ----------
    def list_users(self, **params) -> dict:
        return _get("/users", params=params)

    def create_user(self, data: dict) -> dict:
        return _post("/users", json=data)

    def update_user(self, user_id: int, data: dict) -> dict:
        return _put(f"/users/{user_id}", json=data)

    def disable_user(self, user_id: int) -> dict:
        return _put(f"/users/{user_id}/disable")

    # ---------- 仪表盘 ----------
    def get_stats(self) -> dict:
        return _get("/dashboard/stats")

    def get_top_borrowed(self, limit: int = 10) -> dict:
        return _get("/dashboard/top-borrowed", params={"limit": limit})


# 全局单例
api = ApiClient()

```

```python
"""借用管理页（全部借用记录）"""
from nicegui import ui, app
from api_client import api
from components.navbar import navbar
from components.status_badge import BORROW_STATUS


@ui.page("/borrows")
def borrows_page():
    if not app.storage.user.get("user") or not app.storage.user.get("session_id"):
        ui.navigate.to("/login")
        return

    navbar()
    user = app.storage.user["user"]
    is_admin = user.get("role") == "admin"

    with ui.column().classes("q-pa-md w-full max-w-6xl mx-auto"):
        ui.label("借用管理").classes("text-h5")

        # 筛选
        with ui.row().classes("w-full gap-2 q-mt-md"):
            status_select = ui.select(
                {"": "全部", "active": "借用中", "returned": "已归还", "overdue": "逾期"},
                value="",
            ).classes("w-40").props("outlined dense")
            search_btn = ui.button("筛选", icon="filter_list").props("color=blue")

        table_container = ui.column().classes("w-full q-mt-md")

        def load_borrows(page=1):
            table_container.clear()
            params = {"page": page, "page_size": 20}
            if status_select.value:
                params["status"] = status_select.value

            try:
                result = api.list_borrows(**params)
                if result.get("code") != 0:
                    with table_container:
                        ui.label(f"加载失败: {result.get('msg')}").classes("text-red")
                    return

                data = result["data"]
                records = data.get("items", [])
                total = data.get("total", 0)

                with table_container:
                    if not records:
                        ui.label("暂无借用记录").classes("text-grey text-center q-pa-lg")
                        return

                    columns = [
                        {"name": "item_code", "label": "物品编号", "field": "item_code", "align": "left"},
                        {"name": "item_name", "label": "物品名称", "field": "item_name", "align": "left"},
                        {"name": "user_name", "label": "借用人", "field": "user_name", "align": "left"},
                        {"name": "borrow_date", "label": "借出日期", "field": "borrow_date", "align": "left"},
                        {"name": "due_date", "label": "应还日期", "field": "due_date", "align": "left"},
                        {"name": "status", "label": "状态", "field": "status_label", "align": "center"},
                        {"name": "action", "label": "操作", "field": "action", "align": "center"},
                    ]
                    rows = []
                    for r in records:
                        status_label = BORROW_STATUS.get(r["status"], (r["status"],))[0]
                        rows.append({
                            "id": r["id"],
                            "item_id": r["item_id"],
                            "item_code": r.get("item_code", ""),
                            "item_name": r.get("item_name", ""),
                            "user_name": r.get("user_name", ""),
                            "borrow_date": r.get("borrow_date", "")[:10],
                            "due_date": str(r.get("due_date", "")),
                            "status": r["status"],
                            "status_label": status_label,
                        })

                    table = ui.table(columns=columns, rows=rows, row_key="id").classes("w-full")

                    # 只有 active/overdue 状态且 (本人或管理员) 才能归还
                    table.add_slot("body-cell-action", '''
                        <q-td :props="props">
                            <q-btn v-if="props.row.status === 'active' || props.row.status === 'overdue'"
                                   flat dense color="green" label="归还"
                                   @click="$parent.$emit('return_item', props.row)" />
                            <q-btn flat dense color="blue" icon="visibility"
                                   @click="$parent.$emit('view_item', props.row)" />
                        </q-td>
                    ''')
                    table.on("return_item", lambda e: _show_return_dialog(e.args["id"], load_borrows))
                    table.on("view_item", lambda e: ui.navigate.to(f"/items/{e.args['item_id']}"))

                    # 分页
                    total_pages = (total + 19) // 20
                    if total_pages > 1:
                        with ui.row().classes("w-full justify-center q-mt-md"):
                            for p in range(1, total_pages + 1):
                                btn = ui.button(str(p), on_click=lambda _, pg=p: load_borrows(pg))
                                if p == page:
                                    btn.props("color=blue")
                                else:
                                    btn.props("flat")

            except Exception as e:
                with table_container:
                    ui.label(f"请求失败: {e}").classes("text-red")

        search_btn.on_click(lambda: load_borrows())
        load_borrows()


def _show_return_dialog(borrow_id: int, reload_fn):
    with ui.dialog() as dialog, ui.card().classes("w-96"):
        ui.label("归还物品").classes("text-h6")
        return_note = ui.input("物品状况备注", placeholder="如：完好无损").classes("full-width").props("outlined")
        msg = ui.label("").classes("text-red")

        with ui.row().classes("full-width justify-end gap-2"):
            ui.button("取消", on_click=dialog.close).props("flat")

            def do_return():
                result = api.return_item(borrow_id, return_note.value or "")
                if result.get("code") == 0:
                    dialog.close()
                    ui.notify("归还成功！", type="positive")
                    reload_fn()
                else:
                    msg.text = result.get("msg", "归还失败")

            ui.button("确认归还", on_click=do_return).props("color=green")
    dialog.open()
```

```python
"""物品录入页（Admin）"""
import re
from nicegui import ui, app
from api_client import api
from components.navbar import navbar


def _normalize_date(s: str) -> str | None:
    """把各种日期格式统一为 YYYY-MM-DD"""
    s = s.strip()
    if not s:
        return None
    # 已经是 YYYY-MM-DD
    if re.match(r"^\d{4}-\d{2}-\d{2}$", s):
        return s
    # YYYYMMDD → YYYY-MM-DD
    if re.match(r"^\d{8}$", s):
        return f"{s[:4]}-{s[4:6]}-{s[6:8]}"
    # YYYY/MM/DD → YYYY-MM-DD
    if re.match(r"^\d{4}/\d{2}/\d{2}$", s):
        return s.replace("/", "-")
    return s  # 原样返回，让后端校验


@ui.page("/items/create")
def item_create_page():
    if not app.storage.user.get("user") or not app.storage.user.get("session_id"):
        ui.navigate.to("/login")
        return

    user = app.storage.user["user"]
    if user.get("role") != "admin":
        ui.navigate.to("/items")
        return

    navbar()

    with ui.column().classes("q-pa-md w-full max-w-3xl mx-auto"):
        with ui.row().classes("items-center gap-4"):
```

```python
"""SQLAlchemy 引擎 & Session 管理"""
import os
from sqlalchemy import create_engine, event
from sqlalchemy.orm import sessionmaker, DeclarativeBase

from config import DATABASE_URL, DB_PATH


class Base(DeclarativeBase):
    """ORM 基类"""
    pass


# 确保数据库目录存在
os.makedirs(os.path.dirname(DB_PATH) or ".", exist_ok=True)

# 创建引擎（SQLite 开启 WAL 模式提升并发性能）
engine = create_engine(
    DATABASE_URL,
    connect_args={"check_same_thread": False},
    echo=False,
)


# SQLite 开启外键约束
@event.listens_for(engine, "connect")
def _set_sqlite_pragma(dbapi_conn, connection_record):
    cursor = dbapi_conn.cursor()
    cursor.execute("PRAGMA journal_mode=WAL")
    cursor.execute("PRAGMA foreign_keys=ON")
    cursor.close()


# Session 工厂
SessionLocal = sessionmaker(bind=engine, autocommit=False, autoflush=False)


def get_db():
    """FastAPI 依赖注入：获取数据库 Session"""
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


def init_db():
    """建表（如果不存在）"""
    import models  # noqa: F401 — 触发所有 model 注册
    Base.metadata.create_all(bind=engine)
```

```python
    """每天检查逾期借用"""
    db = SessionLocal()

    scheduler.start()
    print("✓ 逾期检测定时任务已启动 (每天 01:00)")
    yield
    # 关闭时
```

```python
"""认证路由: /api/auth"""
from fastapi import APIRouter, Depends, Response, Request
from sqlalchemy.orm import Session

from database import get_db
from schemas.user import LoginRequest, UserOut
from schemas.common import ApiResponse
from services.user_service import authenticate
from utils.auth import create_session, destroy_session
from utils.deps import get_current_user
from models.user import User

router = APIRouter(prefix="/api/auth", tags=["认证"])


@router.post("/login")
def login(data: LoginRequest, response: Response, db: Session = Depends(get_db)):
    user = authenticate(db, data.username, data.password)
    if not user:
        return ApiResponse(code=40101, msg="用户名或密码错误")

    session_id = create_session(user.id, user.username, user.role, user.real_name)
    response.set_cookie(
        key="session_id",
        value=session_id,
        httponly=True,
        samesite="lax",
        max_age=86400,  # 24小时
    )
    return ApiResponse(data=UserOut.model_validate(user).model_dump())


@router.post("/logout")
def logout(request: Request, response: Response):
    session_id = request.cookies.get("session_id")
    if session_id:
        destroy_session(session_id)
    response.delete_cookie("session_id")
    return ApiResponse(msg="已退出登录")


@router.get("/me")
def get_me(current_user: User = Depends(get_current_user)):
    return ApiResponse(data=UserOut.model_validate(current_user).model_dump())
```

```python
from fastapi import APIRouter, Depends, Query

from schemas.borrow import BorrowCreate, BorrowReturn, BorrowOut
from services import borrow_service

from models.user import User
from models.item import Item

router = APIRouter(prefix="/api/borrows", tags=["借用管理"])


def _borrow_to_out(record, db: Session) -> dict:
    """BorrowRecord ORM → 输出 dict"""
    d = BorrowOut.model_validate(record).model_dump()
    item = db.query(Item).filter_by(id=record.item_id).first()
    user = db.query(User).filter_by(id=record.user_id).first()
    d["item_name"] = item.name if item else ""
    d["item_code"] = item.code if item else ""
    d["user_name"] = user.real_name if user else ""
    return d


@router.post("")
def create_borrow(data: BorrowCreate, db: Session = Depends(get_db), current_user: User = Depends(get_current_user)):
    try:
        record = borrow_service.borrow_item(db, data, current_user)
        return ApiResponse(data=_borrow_to_out(record, db))
    except ValueError as e:
        return ApiResponse(code=40003, msg=str(e))


@router.put("/{borrow_id}/return")
def return_borrow(
    borrow_id: int,
    data: BorrowReturn,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    try:
        record = borrow_service.return_item(db, borrow_id, current_user, data.return_note)
        return ApiResponse(data=_borrow_to_out(record, db))
    except ValueError as e:
        return ApiResponse(code=40003, msg=str(e))


@router.get("")
def list_borrows(

):
    records, total = borrow_service.list_borrows(db, status, page, page_size)

        page_size=page_size,
```

```python
    content = await file.read()
    text = content.decode("utf-8-sig")
    reader = csv.reader(io.StringIO(text))

    created = 0
    errors = []
    for row_num, row in enumerate(reader, 1):
        if row_num == 1 and row[0].startswith("名称"):
            continue  # 跳过表头
        if len(row) < 1:
            continue
        try:
            # CSV: 名称,分类,型号,购入日期,价格,位置,备注
            name = row[0].strip()
            category_name = row[1].strip() if len(row) > 1 else ""
            model_val = row[2].strip() if len(row) > 2 else None
            purchase_date = row[3].strip() if len(row) > 3 else None
            price = row[4].strip() if len(row) > 4 else None
            location = row[5].strip() if len(row) > 5 else None
            remark = row[6].strip() if len(row) > 6 else None

            # 查找分类 ID
            cat_id = None
            if category_name:
                from models.category import Category
                cat = db.query(Category).filter_by(name=category_name).first()
                if cat:
                    cat_id = cat.id

            from decimal import Decimal
            from datetime import date as dt_date
            item_data = ItemCreate(
                name=name,
                category_id=cat_id,
                model=model_val or None,
                purchase_date=dt_date.fromisoformat(purchase_date) if purchase_date else None,
                purchase_price=Decimal(price) if price else None,
                location=location or None,
                remark=remark or None,
            )
            item = item_service.create_item(db, item_data)
            qr_path = generate_qrcode(item.id, item.code)
            item.qrcode_path = qr_path
            db.commit()
            created += 1
        except Exception as e:
            errors.append(f"第{row_num}行: {str(e)}")

    return ApiResponse(data={"created": created, "errors": errors})
```

```python
from datetime import datetime, date, timedelta
from sqlalchemy.orm import Session

from models.borrow import BorrowRecord
from models.item import Item
from models.user import User
from schemas.borrow import BorrowCreate


def borrow_item(db: Session, data: BorrowCreate, user: User) -> BorrowRecord:
    """发起借用"""
    item = db.query(Item).filter_by(id=data.item_id).first()
    if not item:
        raise ValueError("物品不存在")
    if item.status != "available":
        raise ValueError(f"物品当前状态为「{item.status}」，不可借用")

    # 检查 due_date
    if data.due_date < date.today():
        raise ValueError("归还日期不能早于今天")
    if data.due_date > date.today() + timedelta(days=90):
        raise ValueError("借期最长 90 天")

    # 检查用户未归还数量
    active_count = (
        db.query(BorrowRecord)
        .filter_by(user_id=user.id, status="active")
        .count()
    )
    if active_count >= 10:
        raise ValueError("您当前未归还物品已达上限（10件）")

    # 事务：插入借用记录 + 更新物品状态
    record = BorrowRecord(
        item_id=data.item_id,
        user_id=user.id,
        due_date=data.due_date,
        purpose=data.purpose,
        status="active",
    )
    item.status = "borrowed"

    db.add(record)
    db.commit()
    db.refresh(record)
    return record


def return_item(db: Session, borrow_id: int, user: User, return_note: str | None = None) -> BorrowRecord:
    """归还物品"""
    record = db.query(BorrowRecord).filter_by(id=borrow_id).first()
    if not record:
        raise ValueError("借用记录不存在")
    if record.status == "returned":
        raise ValueError("该借用记录已归还")

    # 权限：本人或管理员
    if record.user_id != user.id and user.role != "admin":
        raise ValueError("只能归还自己的借用记录")

    record.status = "returned"
    record.return_date = datetime.now()
    record.return_note = return_note

    # 恢复物品状态（如果物品不是待报废）
    item = db.query(Item).filter_by(id=record.item_id).first()
    if item and item.status == "borrowed":
        item.status = "available"

    db.commit()
    db.refresh(record)
    return record


def list_borrows(
    db: Session,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20,
):
    """所有借用记录"""
    q = db.query(BorrowRecord)
    if status:
        q = q.filter(BorrowRecord.status == status)

    total = q.count()
    records = (
        q.order_by(BorrowRecord.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    return records, total


def my_borrows(
    db: Session,
    user_id: int,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20,
):
    """我的借用记录"""

        q = q.filter(BorrowRecord.status == status)

    records = (

        .limit(page_size)

    )

def list_overdue(db: Session):
    """逾期列表"""
    return (
        db.query(BorrowRecord)
        .filter(
            BorrowRecord.status.in_(["active", "overdue"]),
            BorrowRecord.due_date < date.today(),
        )
        .order_by(BorrowRecord.due_date)
        .all()
```

```python
    borrowed = db.query(Item).filter_by(status="borrowed").count()
    scrapped = db.query(Item).filter_by(status="scrapped").count()

    overdue = (

        "scrapped_count": scrapped,
    }


def get_top_borrowed(db: Session, limit: int = 10) -> list:
    """借用频次 Top N"""
    results = (
        db.query(
            BorrowRecord.item_id,
            Item.name,
            Item.code,
            func.count(BorrowRecord.id).label("borrow_count"),
        )
        .join(Item, BorrowRecord.item_id == Item.id)
        .group_by(BorrowRecord.item_id)
        .order_by(func.count(BorrowRecord.id).desc())
        .limit(limit)
        .all()
    )
    return [
        {
            "item_id": r.item_id,
            "name": r.name,
            "code": r.code,
            "borrow_count": r.borrow_count,
        }
        for r in results
    ]


def export_borrow_report(db: Session, start_date: date | None = None, end_date: date | None = None) -> str:

    if start_date:
        q = q.filter(BorrowRecord.borrow_date >= start_date)
    if end_date:

    writer = csv.writer(output)

            str(r.due_date) if r.due_date else "",
            r.return_date.strftime("%Y-%m-%d") if r.return_date else "未归还",
            r.status,
            r.purpose or "",
        ])
```

```python
"""物品业务逻辑"""
from datetime import date
from sqlalchemy.orm import Session
from sqlalchemy import func

from models.item import Item
from models.category import Category
from schemas.item import ItemCreate, ItemUpdate


def generate_item_code(db: Session) -> str:
    """生成物品编号: ITM-YYYYMMDD-XXXX"""
    today = date.today().strftime("%Y%m%d")
    prefix = f"ITM-{today}-"

    # 查询当天最大编号
    last = (
        db.query(Item.code)
        .filter(Item.code.like(f"{prefix}%"))
        .order_by(Item.code.desc())
        .first()
    )

    if last:
        seq = int(last[0].split("-")[-1]) + 1
    else:
        seq = 1

    return f"{prefix}{seq:04d}"


def create_item(db: Session, data: ItemCreate) -> Item:
    """录入物品"""
    # 校验分类存在
    if data.category_id:
        cat = db.query(Category).filter_by(id=data.category_id).first()
        if not cat:
            raise ValueError("分类不存在")

    code = generate_item_code(db)
    item = Item(
        code=code,
        name=data.name,
        category_id=data.category_id,
        model=data.model,
        status="available",
        purchase_date=data.purchase_date,
        purchase_price=data.purchase_price,
        location=data.location,
        remark=data.remark,
    )
    db.add(item)
    db.commit()
    db.refresh(item)
    return item


def get_item(db: Session, item_id: int) -> Item | None:
    """获取单个物品"""
    return db.query(Item).filter_by(id=item_id).first()


def list_items(
    db: Session,
    keyword: str | None = None,
    category_id: int | None = None,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20,
):
    """分页查询物品列表"""
    q = db.query(Item)

    if keyword:
        q = q.filter(
            (Item.name.contains(keyword)) | (Item.code.contains(keyword))
        )
    if category_id:
        q = q.filter(Item.category_id == category_id)
    if status:
        q = q.filter(Item.status == status)

    total = q.count()
    items = (
        q.order_by(Item.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
```

```python
import os
import qrcode
from datetime import datetime
from reportlab.lib.pagesizes import A4
from reportlab.pdfgen import canvas

from config import SERVER_HOST, QRCODE_DIR, EXPORT_DIR


def generate_qrcode(item_id: int, item_code: str) -> str:
    """生成二维码图片，返回保存路径"""
    os.makedirs(QRCODE_DIR, exist_ok=True)

    url = f"http://{SERVER_HOST}/items/{item_id}"

    qr = qrcode.QRCode(version=1, box_size=10, border=2)
    qr.add_data(url)
    qr.make(fit=True)
    img = qr.make_image(fill_color="black", back_color="white")

    path = os.path.join(QRCODE_DIR, f"{item_code}.png")
    img.save(path)
    return path


def generate_qrcode_pdf(items: list) -> str:
    """生成包含多个物品二维码的 PDF
    
    items: list of dicts with keys: qrcode_path, name, code
    """
    os.makedirs(EXPORT_DIR, exist_ok=True)
    timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
    pdf_path = os.path.join(EXPORT_DIR, f"qrcodes_{timestamp}.pdf")

    c = canvas.Canvas(pdf_path, pagesize=A4)
    width, height = A4
    cols, rows = 2, 4  # 每页 2列 x 4行
    cell_w = width / cols
    cell_h = height / rows

    for i, item in enumerate(items):
        page_idx = i % (cols * rows)
        if page_idx == 0 and i > 0:
            c.showPage()

        col = page_idx % cols
        row = page_idx // cols

        x = col * cell_w + 20
        y = height - (row + 1) * cell_h + 20

        # 画二维码图片
        qr_path = item.get("qrcode_path") or ""
        if os.path.exists(qr_path):
            c.drawImage(qr_path, x, y + 30, 80, 80)

        # 物品名称
        c.setFont("Helvetica", 10)
        c.drawString(x, y + 15, item.get("name", ""))
        # 物品编号
        c.setFont("Helvetica", 8)
        c.drawString(x, y, item.get("code", ""))

    c.save()
    return pdf_path
```

```python
"""报废业务逻辑"""
from datetime import datetime
from sqlalchemy.orm import Session

from models.scrap import ScrapRecord
from models.item import Item
from models.user import User


def apply_scrap(db: Session, item_id: int, reason: str, applicant: User) -> ScrapRecord:
    """发起报废申请"""
    item = db.query(Item).filter_by(id=item_id).first()
    if not item:
        raise ValueError("物品不存在")
    if item.status == "scrapped":
        raise ValueError("物品已报废")
    if item.status == "pending_scrap":
        raise ValueError("物品已有待审批的报废申请")

    # 检查是否已有 pending 的报废申请
    existing = (
        db.query(ScrapRecord)
        .filter_by(item_id=item_id, status="pending")
        .first()
    )
    if existing:
        raise ValueError("该物品已有待审批的报废申请")

    record = ScrapRecord(
        item_id=item_id,
        applicant_id=applicant.id,
        reason=reason,
        status="pending",
    )
    item.status = "pending_scrap"

    db.add(record)
    db.commit()
    db.refresh(record)
    return record


def approve_scrap(db: Session, scrap_id: int, approver: User, approve_note: str | None = None) -> ScrapRecord:
    """审批通过报废"""
    record = db.query(ScrapRecord).filter_by(id=scrap_id).first()
    if not record:
        raise ValueError("报废记录不存在")
    if record.status != "pending":
        raise ValueError("该申请不在待审批状态")

    record.status = "approved"
    record.approver_id = approver.id
    record.approve_time = datetime.now()
    record.approve_note = approve_note

    # 物品状态 → scrapped
    item = db.query(Item).filter_by(id=record.item_id).first()
    if item:
        item.status = "scrapped"

    db.commit()
    db.refresh(record)
    return record


def reject_scrap(db: Session, scrap_id: int, approver: User, approve_note: str | None = None) -> ScrapRecord:
    """审批驳回报废"""
    record = db.query(ScrapRecord).filter_by(id=scrap_id).first()
    if not record:
        raise ValueError("报废记录不存在")
    if record.status != "pending":
        raise ValueError("该申请不在待审批状态")

    record.status = "rejected"
    record.approver_id = approver.id
    record.approve_time = datetime.now()
    record.approve_note = approve_note

    # 恢复物品状态
    item = db.query(Item).filter_by(id=record.item_id).first()
    if item and item.status == "pending_scrap":
        # 检查是否有活跃借用记录来决定恢复什么状态
        from models.borrow import BorrowRecord
        active_borrow = (
            db.query(BorrowRecord)
            .filter_by(item_id=item.id, status="active")
            .first()
        )
        item.status = "borrowed" if active_borrow else "available"

    db.commit()
    db.refresh(record)
    return record


def list_scraps(
    db: Session,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20,
):
    """报废申请列表"""
    q = db.query(ScrapRecord)
    if status:
        q = q.filter(ScrapRecord.status == status)

    total = q.count()
    records = (
        q.order_by(ScrapRecord.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    return records, total
```

```python
"""Session 认证工具 — 基于 Cookie 的 Session 管理"""
import hashlib
import uuid
from datetime import datetime


def hash_password(password: str) -> str:
    """密码哈希 (SHA256)"""
    return hashlib.sha256(password.encode()).hexdigest()


def verify_password(plain: str, hashed: str) -> bool:
    """验证密码"""
    return hash_password(plain) == hashed


def generate_session_id() -> str:
    """生成随机 Session ID"""
    return uuid.uuid4().hex


# 内存 Session 存储 — 单机够用，重启后需重新登录
# 格式: {session_id: {"user_id": int, "username": str, "role": str, "created": datetime}}
_sessions: dict[str, dict] = {}


def create_session(user_id: int, username: str, role: str, real_name: str) -> str:
    """创建 Session，返回 session_id"""
    sid = generate_session_id()
    _sessions[sid] = {
        "user_id": user_id,
        "username": username,
        "role": role,
        "real_name": real_name,
        "created": datetime.now(),
    }
    return sid


def get_session(session_id: str) -> dict | None:
    """根据 session_id 获取 Session 数据"""
    return _sessions.get(session_id)


def destroy_session(session_id: str):
    """销毁 Session"""
    _sessions.pop(session_id, None)
```

```python
"""FastAPI 依赖注入"""
from fastapi import Request, Depends, HTTPException
from sqlalchemy.orm import Session

from database import get_db
from utils.auth import get_session
from models.user import User


def get_current_user(request: Request, db: Session = Depends(get_db)) -> User:
    """从 Cookie 中读取 session_id，获取当前登录用户"""
    session_id = request.cookies.get("session_id")
    if not session_id:
        raise HTTPException(status_code=401, detail="未登录")

    session_data = get_session(session_id)
    if not session_data:
        raise HTTPException(status_code=401, detail="Session 已过期，请重新登录")

    user = db.query(User).filter_by(id=session_data["user_id"]).first()
    if not user or not user.is_active:
        raise HTTPException(status_code=401, detail="用户不存在或已禁用")

    return user


def require_admin(current_user: User = Depends(get_current_user)) -> User:
    """要求管理员权限"""
    if current_user.role != "admin":
        raise HTTPException(status_code=403, detail="权限不足，需要管理员权限")
    return current_user
```

"""认证路由: /api/auth"""
from fastapi import APIRouter, Depends, Response, Request
from sqlalchemy.orm import Session

from database import get_db
from schemas.user import LoginRequest, UserOut
from schemas.common import ApiResponse
from services.user_service import authenticate
from utils.auth import create_session, destroy_session
from utils.deps import get_current_user
from models.user import User

router = APIRouter(prefix="/api/auth", tags=["认证"])


@router.post("/login")
def login(data: LoginRequest, response: Response, db: Session = Depends(get_db)):
    user = authenticate(db, data.username, data.password)
    if not user:
        return ApiResponse(code=40101, msg="用户名或密码错误")

    session_id = create_session(user.id, user.username, user.role, user.real_name)
    response.set_cookie(
        key="session_id",
        value=session_id,
        httponly=True,
        samesite="lax",
        max_age=86400,  # 24小时
    )
    return ApiResponse(data=UserOut.model_validate(user).model_dump())


@router.post("/logout")
def logout(request: Request, response: Response):
    session_id = request.cookies.get("session_id")
    if session_id:
        destroy_session(session_id)
    response.delete_cookie("session_id")
    return ApiResponse(msg="已退出登录")


@router.get("/me")
def get_me(current_user: User = Depends(get_current_user)):
    return ApiResponse(data=UserOut.model_validate(current_user).model_dump())

router = APIRouter(prefix="/api/borrows", tags=["借用管理"])


def _borrow_to_out(record, db: Session) -> dict:
    """BorrowRecord ORM → 输出 dict"""
    d = BorrowOut.model_validate(record).model_dump()
    item = db.query(Item).filter_by(id=record.item_id).first()
    user = db.query(User).filter_by(id=record.user_id).first()
    d["item_name"] = item.name if item else ""
    d["item_code"] = item.code if item else ""
    d["user_name"] = user.real_name if user else ""
    return d

        record = borrow_service.borrow_item(db, data, current_user)
        return ApiResponse(data=_borrow_to_out(record, db))
    except ValueError as e:
        return ApiResponse(code=40003, msg=str(e))

@router.get("/{item_id}")
def get_item(item_id: int, db: Session = Depends(get_db)):
    item = item_service.get_item(db, item_id)
    if not item:
        return ApiResponse(code=40001, msg="物品不存在")

    data = _item_to_out(item, db)
    # 附加借用历史
    history = get_borrow_history(db, item_id, limit=5)
    from models.user import User as UserModel
    data["borrow_history"] = []
    for h in history:
        u = db.query(UserModel).filter_by(id=h.user_id).first()
        data["borrow_history"].append({
            "id": h.id,
            "user_name": u.real_name if u else "",
            "borrow_date": h.borrow_date.isoformat() if h.borrow_date else "",
            "due_date": str(h.due_date) if h.due_date else "",
            "return_date": h.return_date.isoformat() if h.return_date else None,
            "status": h.status,
        })
    return ApiResponse(data=data)

@router.get("/{item_id}/qrcode")

    item = item_service.get_item(db, item_id)
    if not item:
        return ApiResponse(code=40001, msg="物品不存在")
    if not item.qrcode_path:
        # 按需生成
        qr_path = generate_qrcode(item.id, item.code)
        item.qrcode_path = qr_path
        db.commit()
    return FileResponse(item.qrcode_path, media_type="image/png")


@router.post("/qrcode-pdf")
def batch_qrcode_pdf(data: QrcodePdfRequest, db: Session = Depends(get_db), admin: User = Depends(require_admin)):
    items = []
    for iid in data.item_ids:
        item = item_service.get_item(db, iid)
        if item:
            if not item.qrcode_path:
                item.qrcode_path = generate_qrcode(item.id, item.code)
                db.commit()
            items.append({"qrcode_path": item.qrcode_path, "name": item.name, "code": item.code})

    if not items:
        return ApiResponse(code=40001, msg="未找到有效物品")

    pdf_path = generate_qrcode_pdf(items)
    return FileResponse(pdf_path, media_type="application/pdf", filename="qrcodes.pdf")


@router.post("/import")
async def import_csv(file: UploadFile = File(...), db: Session = Depends(get_db), admin: User = Depends(require_admin)):
    """CSV 批量导入物品"""
    content = await file.read()
    text = content.decode("utf-8-sig")
    reader = csv.reader(io.StringIO(text))

    created = 0
    errors = []
    for row_num, row in enumerate(reader, 1):
        if row_num == 1 and row[0].startswith("名称"):
            continue  # 跳过表头
        if len(row) < 1:
            continue
        try:
            # CSV: 名称,分类,型号,购入日期,价格,位置,备注
            name = row[0].strip()
            category_name = row[1].strip() if len(row) > 1 else ""
            model_val = row[2].strip() if len(row) > 2 else None
            purchase_date = row[3].strip() if len(row) > 3 else None
            price = row[4].strip() if len(row) > 4 else None
            location = row[5].strip() if len(row) > 5 else None
            remark = row[6].strip() if len(row) > 6 else None

            # 查找分类 ID
            cat_id = None
            if category_name:
                from models.category import Category
                cat = db.query(Category).filter_by(name=category_name).first()
                if cat:
                    cat_id = cat.id

            from decimal import Decimal
            from datetime import date as dt_date
            item_data = ItemCreate(
                name=name,
                category_id=cat_id,
                model=model_val or None,
                purchase_date=dt_date.fromisoformat(purchase_date) if purchase_date else None,
                purchase_price=Decimal(price) if price else None,
                location=location or None,
                remark=remark or None,
            )
            item = item_service.create_item(db, item_data)
            qr_path = generate_qrcode(item.id, item.code)
            item.qrcode_path = qr_path
            db.commit()
            created += 1
        except Exception as e:
            errors.append(f"第{row_num}行: {str(e)}")

    return ApiResponse(data={"created": created, "errors": errors})

from datetime import datetime, date, timedelta
from sqlalchemy.orm import Session

from models.borrow import BorrowRecord
from models.item import Item
from models.user import User
from schemas.borrow import BorrowCreate


def borrow_item(db: Session, data: BorrowCreate, user: User) -> BorrowRecord:
    """发起借用"""
    item = db.query(Item).filter_by(id=data.item_id).first()
    if not item:
        raise ValueError("物品不存在")
    if item.status != "available":
        raise ValueError(f"物品当前状态为「{item.status}」，不可借用")

    # 检查 due_date
    if data.due_date < date.today():
        raise ValueError("归还日期不能早于今天")
    if data.due_date > date.today() + timedelta(days=90):
        raise ValueError("借期最长 90 天")

    # 检查用户未归还数量
    active_count = (
        db.query(BorrowRecord)
        .filter_by(user_id=user.id, status="active")
        .count()
    )
    if active_count >= 10:
        raise ValueError("您当前未归还物品已达上限（10件）")

    # 事务：插入借用记录 + 更新物品状态
    record = BorrowRecord(
        item_id=data.item_id,
        user_id=user.id,
        due_date=data.due_date,
        purpose=data.purpose,
        status="active",
    )
    item.status = "borrowed"

    db.add(record)
    db.commit()
    db.refresh(record)
    return record


def return_item(db: Session, borrow_id: int, user: User, return_note: str | None = None) -> BorrowRecord:
    """归还物品"""
    record = db.query(BorrowRecord).filter_by(id=borrow_id).first()
    if not record:
        raise ValueError("借用记录不存在")
    if record.status == "returned":
        raise ValueError("该借用记录已归还")

    # 权限：本人或管理员
    if record.user_id != user.id and user.role != "admin":
        raise ValueError("只能归还自己的借用记录")

    record.status = "returned"
    record.return_date = datetime.now()
    record.return_note = return_note

    # 恢复物品状态（如果物品不是待报废）
    item = db.query(Item).filter_by(id=record.item_id).first()
    if item and item.status == "borrowed":
        item.status = "available"

    db.commit()
    db.refresh(record)
    return record


def list_borrows(
    db: Session,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20,
):
    """所有借用记录"""
    q = db.query(BorrowRecord)
    if status:
        q = q.filter(BorrowRecord.status == status)

    total = q.count()
    records = (
        q.order_by(BorrowRecord.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    return records, total

import csv

def get_stats(db: Session) -> dict:
    """统计概览"""
    total = db.query(Item).count()
    available = db.query(Item).filter_by(status="available").count()
    borrowed = db.query(Item).filter_by(status="borrowed").count()
    scrapped = db.query(Item).filter_by(status="scrapped").count()

    overdue = (
        db.query(BorrowRecord)
        .filter(
            BorrowRecord.status.in_(["active", "overdue"]),
            BorrowRecord.due_date < date.today(),
        )
        .count()
    )

    return {
        "total_items": total,
        "available_count": available,
        "borrowed_count": borrowed,
        "overdue_count": overdue,
        "scrapped_count": scrapped,
    }


def get_top_borrowed(db: Session, limit: int = 10) -> list:
    """借用频次 Top N"""
    results = (
        db.query(
            BorrowRecord.item_id,
            Item.name,
            Item.code,
            func.count(BorrowRecord.id).label("borrow_count"),
        )
        .join(Item, BorrowRecord.item_id == Item.id)
        .group_by(BorrowRecord.item_id)
        .order_by(func.count(BorrowRecord.id).desc())
        .limit(limit)
        .all()
    )
    return [
        {
            "item_id": r.item_id,
            "name": r.name,
            "code": r.code,
            "borrow_count": r.borrow_count,
        }
        for r in results
    ]


def export_borrow_report(db: Session, start_date: date | None = None, end_date: date | None = None) -> str:
    """导出借用报表 CSV，返回 CSV 字符串"""
    q = db.query(BorrowRecord).join(Item, BorrowRecord.item_id == Item.id).join(User, BorrowRecord.user_id == User.id)

    if start_date:
        q = q.filter(BorrowRecord.borrow_date >= start_date)
    if end_date:
        q = q.filter(BorrowRecord.borrow_date <= end_date)

    records = q.order_by(BorrowRecord.borrow_date.desc()).all()

    output = io.StringIO()
    writer = csv.writer(output)
    writer.writerow(["借用ID", "物品编号", "物品名称", "借用人", "借出日期", "应还日期", "实际归还", "状态", "用途"])

    for r in records:
        item = db.query(Item).filter_by(id=r.item_id).first()
        user = db.query(User).filter_by(id=r.user_id).first()
        writer.writerow([
            r.id,
            item.code if item else "",
            item.name if item else "",
            user.real_name if user else "",
            r.borrow_date.strftime("%Y-%m-%d") if r.borrow_date else "",
            str(r.due_date) if r.due_date else "",
            r.return_date.strftime("%Y-%m-%d") if r.return_date else "未归还",
            r.status,
            r.purpose or "",
        ])

    return output.getvalue()

from datetime import date
from sqlalchemy.orm import Session
from sqlalchemy import func

def generate_item_code(db: Session) -> str:
    """生成物品编号: ITM-YYYYMMDD-XXXX"""
    today = date.today().strftime("%Y%m%d")
    prefix = f"ITM-{today}-"

    # 查询当天最大编号
    last = (
        db.query(Item.code)
        .filter(Item.code.like(f"{prefix}%"))
        .order_by(Item.code.desc())
        .first()
    )

    if last:
        seq = int(last[0].split("-")[-1]) + 1
    else:
        seq = 1

    return f"{prefix}{seq:04d}"


def create_item(db: Session, data: ItemCreate) -> Item:
    """录入物品"""
    # 校验分类存在
    if data.category_id:
        cat = db.query(Category).filter_by(id=data.category_id).first()
        if not cat:
            raise ValueError("分类不存在")

    code = generate_item_code(db)
    item = Item(
        code=code,
        name=data.name,
        category_id=data.category_id,
        model=data.model,
        status="available",
        purchase_date=data.purchase_date,
        purchase_price=data.purchase_price,
        location=data.location,
        remark=data.remark,
    )
    db.add(item)
    db.commit()
    db.refresh(item)
    return item

    page_size: int = 20,

    if status:

    total = q.count()
    items = (
        q.order_by(Item.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    return items, total

"""二维码生成 & PDF 导出"""
import os
import qrcode
from datetime import datetime
from reportlab.lib.pagesizes import A4
from reportlab.pdfgen import canvas

from config import SERVER_HOST, QRCODE_DIR, EXPORT_DIR


def generate_qrcode(item_id: int, item_code: str) -> str:
    """生成二维码图片，返回保存路径"""
    os.makedirs(QRCODE_DIR, exist_ok=True)

    url = f"http://{SERVER_HOST}/items/{item_id}"

    qr = qrcode.QRCode(version=1, box_size=10, border=2)
    qr.add_data(url)
    qr.make(fit=True)
    img = qr.make_image(fill_color="black", back_color="white")

    path = os.path.join(QRCODE_DIR, f"{item_code}.png")
    img.save(path)
    return path


def generate_qrcode_pdf(items: list) -> str:
    """生成包含多个物品二维码的 PDF
    
    items: list of dicts with keys: qrcode_path, name, code
    """
    os.makedirs(EXPORT_DIR, exist_ok=True)
    timestamp = datetime.now().strftime("%Y%m%d%H%M%S")
    pdf_path = os.path.join(EXPORT_DIR, f"qrcodes_{timestamp}.pdf")

    c = canvas.Canvas(pdf_path, pagesize=A4)
    width, height = A4
    cols, rows = 2, 4  # 每页 2列 x 4行
    cell_w = width / cols
    cell_h = height / rows

    for i, item in enumerate(items):
        page_idx = i % (cols * rows)
        if page_idx == 0 and i > 0:
            c.showPage()

        col = page_idx % cols
        row = page_idx // cols

        x = col * cell_w + 20
        y = height - (row + 1) * cell_h + 20

        # 画二维码图片
        qr_path = item.get("qrcode_path") or ""
        if os.path.exists(qr_path):
            c.drawImage(qr_path, x, y + 30, 80, 80)

        # 物品名称
        c.setFont("Helvetica", 10)
        c.drawString(x, y + 15, item.get("name", ""))
        # 物品编号
        c.setFont("Helvetica", 8)
        c.drawString(x, y, item.get("code", ""))

    c.save()
    return pdf_path

"""报废业务逻辑"""
from datetime import datetime
from sqlalchemy.orm import Session

from models.scrap import ScrapRecord
from models.item import Item
from models.user import User


def apply_scrap(db: Session, item_id: int, reason: str, applicant: User) -> ScrapRecord:
    """发起报废申请"""
    item = db.query(Item).filter_by(id=item_id).first()
    if not item:
        raise ValueError("物品不存在")
    if item.status == "scrapped":
        raise ValueError("物品已报废")
    if item.status == "pending_scrap":
        raise ValueError("物品已有待审批的报废申请")

    # 检查是否已有 pending 的报废申请
    existing = (
        db.query(ScrapRecord)
        .filter_by(item_id=item_id, status="pending")
        .first()
    )
    if existing:
        raise ValueError("该物品已有待审批的报废申请")

    record = ScrapRecord(
        item_id=item_id,
        applicant_id=applicant.id,
        reason=reason,
        status="pending",
    )
    item.status = "pending_scrap"

    db.add(record)
    db.commit()
    db.refresh(record)
    return record


def approve_scrap(db: Session, scrap_id: int, approver: User, approve_note: str | None = None) -> ScrapRecord:
    """审批通过报废"""
    record = db.query(ScrapRecord).filter_by(id=scrap_id).first()
    if not record:
        raise ValueError("报废记录不存在")
    if record.status != "pending":
        raise ValueError("该申请不在待审批状态")

    record.status = "approved"
    record.approver_id = approver.id
    record.approve_time = datetime.now()
    record.approve_note = approve_note

    # 物品状态 → scrapped
    item = db.query(Item).filter_by(id=record.item_id).first()
    if item:
        item.status = "scrapped"

    db.commit()
    db.refresh(record)
    return record


def reject_scrap(db: Session, scrap_id: int, approver: User, approve_note: str | None = None) -> ScrapRecord:
    """审批驳回报废"""
    record = db.query(ScrapRecord).filter_by(id=scrap_id).first()
    if not record:
        raise ValueError("报废记录不存在")
    if record.status != "pending":
        raise ValueError("该申请不在待审批状态")

    record.status = "rejected"
    record.approver_id = approver.id
    record.approve_time = datetime.now()
    record.approve_note = approve_note

    # 恢复物品状态
    item = db.query(Item).filter_by(id=record.item_id).first()
    if item and item.status == "pending_scrap":
        # 检查是否有活跃借用记录来决定恢复什么状态
        from models.borrow import BorrowRecord
        active_borrow = (
            db.query(BorrowRecord)
            .filter_by(item_id=item.id, status="active")
            .first()
        )
        item.status = "borrowed" if active_borrow else "available"

    db.commit()
    db.refresh(record)
    return record


def list_scraps(
    db: Session,
    status: str | None = None,
    page: int = 1,
    page_size: int = 20,
):
    """报废申请列表"""
    q = db.query(ScrapRecord)
    if status:
        q = q.filter(ScrapRecord.status == status)

    total = q.count()
    records = (
        q.order_by(ScrapRecord.id.desc())
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    return records, total

"""用户业务逻辑"""
from sqlalchemy.orm import Session

from models.user import User
from schemas.user import UserCreate, UserUpdate
from utils.auth import hash_password, verify_password


def authenticate(db: Session, username: str, password: str) -> User | None:
    """验证用户名密码，成功返回 User，失败返回 None"""
    user = db.query(User).filter_by(username=username).first()
    if not user or not user.is_active:
        return None
    if not verify_password(password, user.password_hash):
        return None
    return user


def create_user(db: Session, data: UserCreate) -> User:
    """创建用户"""
    existing = db.query(User).filter_by(username=data.username).first()
    if existing:
        raise ValueError(f"用户名 '{data.username}' 已存在")

    user = User(
        username=data.username,
        password_hash=hash_password(data.password),
        real_name=data.real_name,
        role=data.role,
        phone=data.phone,
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    return user


def update_user(db: Session, user_id: int, data: UserUpdate) -> User:
    """更新用户信息"""
    user = db.query(User).filter_by(id=user_id).first()
    if not user:
        raise ValueError("用户不存在")

    for field, value in data.model_dump(exclude_unset=True).items():
        setattr(user, field, value)

    db.commit()
    db.refresh(user)
    return user


def list_users(db: Session, page: int = 1, page_size: int = 20):
    """分页查询用户列表"""
    total = db.query(User).count()
    users = (
        db.query(User)
        .order_by(User.id)
        .offset((page - 1) * page_size)
        .limit(page_size)
        .all()
    )
    return users, total


def disable_user(db: Session, user_id: int) -> User:
    """禁用用户"""
    user = db.query(User).filter_by(id=user_id).first()
    if not user:
        raise ValueError("用户不存在")
    user.is_active = False
    db.commit()
    db.refresh(user)
    return user

"""Session 认证工具 — 基于 Cookie 的 Session 管理"""
import hashlib
import uuid
from datetime import datetime


def hash_password(password: str) -> str:
    """密码哈希 (SHA256)"""
    return hashlib.sha256(password.encode()).hexdigest()


def verify_password(plain: str, hashed: str) -> bool:
    """验证密码"""
    return hash_password(plain) == hashed


def generate_session_id() -> str:
    """生成随机 Session ID"""
    return uuid.uuid4().hex


# 内存 Session 存储 — 单机够用，重启后需重新登录
# 格式: {session_id: {"user_id": int, "username": str, "role": str, "created": datetime}}
_sessions: dict[str, dict] = {}


def create_session(user_id: int, username: str, role: str, real_name: str) -> str:
    """创建 Session，返回 session_id"""
    sid = generate_session_id()
    _sessions[sid] = {
        "user_id": user_id,
        "username": username,
        "role": role,
        "real_name": real_name,
        "created": datetime.now(),
    }
    return sid


def get_session(session_id: str) -> dict | None:
    """根据 session_id 获取 Session 数据"""
    return _sessions.get(session_id)


def destroy_session(session_id: str):
    """销毁 Session"""
    _sessions.pop(session_id, None)

"""FastAPI 依赖注入"""
from fastapi import Request, Depends, HTTPException
from sqlalchemy.orm import Session

from database import get_db
from utils.auth import get_session
from models.user import User


def get_current_user(request: Request, db: Session = Depends(get_db)) -> User:
    """从 Cookie 中读取 session_id，获取当前登录用户"""
    session_id = request.cookies.get("session_id")
    if not session_id:
        raise HTTPException(status_code=401, detail="未登录")

    session_data = get_session(session_id)
    if not session_data:
        raise HTTPException(status_code=401, detail="Session 已过期，请重新登录")

    user = db.query(User).filter_by(id=session_data["user_id"]).first()
    if not user or not user.is_active:
        raise HTTPException(status_code=401, detail="用户不存在或已禁用")

    return user


def require_admin(current_user: User = Depends(get_current_user)) -> User:
    """要求管理员权限"""
    if current_user.role != "admin":
        raise HTTPException(status_code=403, detail="权限不足，需要管理员权限")
    return current_user
