"""封装后端 API 调用 — 每次请求从 NiceGUI app.storage.user 读 session_id"""
import os
import httpx
from nicegui import app

API_BASE = os.getenv("API_BASE_URL", "http://localhost:8000/api")


def _get_session_id() -> str | None:
    """从 NiceGUI 用户存储读取 session_id"""
    try:
        return app.storage.user.get("session_id")
    except Exception:
        return None


def _cookies() -> dict:
    """构造 Cookie dict"""
    sid = _get_session_id()
    return {"session_id": sid} if sid else {}


def _post(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.post(path, cookies=_cookies(), **kwargs)
        return resp.json()


def _get(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.get(path, cookies=_cookies(), **kwargs)
        return resp.json()


def _put(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.put(path, cookies=_cookies(), **kwargs)
        return resp.json()


def _delete(path: str, **kwargs) -> dict:
    with httpx.Client(base_url=API_BASE, timeout=30) as c:
        resp = c.delete(path, cookies=_cookies(), **kwargs)
        return resp.json()


class ApiClient:

    # ---------- 认证 ----------
    def login(self, username: str, password: str) -> dict:
        """登录 — 特殊处理：需要从 response 提取 session_id cookie"""
        with httpx.Client(base_url=API_BASE, timeout=30) as c:
            resp = c.post("/auth/login", json={"username": username, "password": password})
            data = resp.json()
            # 把后端返回的 session_id cookie 存到 NiceGUI user storage
            sid = resp.cookies.get("session_id")
            if sid:
                app.storage.user["session_id"] = sid
            return data

    def logout(self) -> dict:
        result = _post("/auth/logout")
        try:
            app.storage.user.pop("session_id", None)
        except Exception:
            pass
        return result

    def get_me(self) -> dict:
        return _get("/auth/me")

    # ---------- 物品 ----------
    def list_items(self, **params) -> dict:
        return _get("/items", params=params)

    def get_item(self, item_id: int) -> dict:
        return _get(f"/items/{item_id}")

    def create_item(self, data: dict) -> dict:
        return _post("/items", json=data)

    def update_item(self, item_id: int, data: dict) -> dict:
        return _put(f"/items/{item_id}", json=data)

    def delete_item(self, item_id: int) -> dict:
        return _delete(f"/items/{item_id}")

    def get_categories(self) -> dict:
        return _get("/items/categories")

    def get_qrcode_url(self, item_id: int) -> str:
        """返回二维码图片的完整 URL（含 session cookie 无法直接用，走代理）"""
        return f"{API_BASE.replace('/api', '')}/api/items/{item_id}/qrcode"

    # ---------- 借用 ----------
    def borrow_item(self, item_id: int, due_date: str, purpose: str = "") -> dict:
        return _post("/borrows", json={
            "item_id": item_id, "due_date": due_date, "purpose": purpose,
        })

    def return_item(self, borrow_id: int, return_note: str = "") -> dict:
        return _put(f"/borrows/{borrow_id}/return", json={"return_note": return_note})

    def list_borrows(self, **params) -> dict:
        return _get("/borrows", params=params)

    def my_borrows(self, **params) -> dict:
        return _get("/borrows/my", params=params)

    def list_overdue(self) -> dict:
        return _get("/borrows/overdue")

    # ---------- 报废 ----------
    def apply_scrap(self, item_id: int, reason: str) -> dict:
        return _post("/scraps", json={"item_id": item_id, "reason": reason})

    def approve_scrap(self, scrap_id: int, note: str = "") -> dict:
        return _put(f"/scraps/{scrap_id}/approve", json={"approve_note": note})

    def reject_scrap(self, scrap_id: int, note: str = "") -> dict:
        return _put(f"/scraps/{scrap_id}/reject", json={"approve_note": note})

    def list_scraps(self, **params) -> dict:
        return _get("/scraps", params=params)

    # ---------- 用户 ----------
    def list_users(self, **params) -> dict:
        return _get("/users", params=params)

    def create_user(self, data: dict) -> dict:
        return _post("/users", json=data)

    def update_user(self, user_id: int, data: dict) -> dict:
        return _put(f"/users/{user_id}", json=data)

    def disable_user(self, user_id: int) -> dict:
        return _put(f"/users/{user_id}/disable")

    # ---------- 仪表盘 ----------
    def get_stats(self) -> dict:
        return _get("/dashboard/stats")

    def get_top_borrowed(self, limit: int = 10) -> dict:
        return _get("/dashboard/top-borrowed", params={"limit": limit})


# 全局单例
api = ApiClient()

def navbar():

from nicegui import ui, app
from api_client import api
from components.navbar import navbar

@ui.page("/borrows")
def borrows_page():
    if not app.storage.user.get("user") or not app.storage.user.get("session_id"):
        ui.navigate.to("/login")
        return

    navbar()
    user = app.storage.user["user"]
    is_admin = user.get("role") == "admin"

    with ui.column().classes("q-pa-md w-full max-w-6xl mx-auto"):
        ui.label("借用管理").classes("text-h5")

        # 筛选
        with ui.row().classes("w-full gap-2 q-mt-md"):
            status_select = ui.select(
                {"": "全部", "active": "借用中", "returned": "已归还", "overdue": "逾期"},
                value="",
            ).classes("w-40").props("outlined dense")
            search_btn = ui.button("筛选", icon="filter_list").props("color=blue")

        table_container = ui.column().classes("w-full q-mt-md")

        def load_borrows(page=1):
            table_container.clear()
            params = {"page": page, "page_size": 20}
            if status_select.value:
                params["status"] = status_select.value

            try:
                result = api.list_borrows(**params)
                if result.get("code") != 0:
                    with table_container:
                        ui.label(f"加载失败: {result.get('msg')}").classes("text-red")
                    return

                data = result["data"]
                records = data.get("items", [])
                total = data.get("total", 0)

                with table_container:
                    if not records:
                        ui.label("暂无借用记录").classes("text-grey text-center q-pa-lg")
                        return

                    columns = [
                        {"name": "item_code", "label": "物品编号", "field": "item_code", "align": "left"},
                        {"name": "item_name", "label": "物品名称", "field": "item_name", "align": "left"},
                        {"name": "user_name", "label": "借用人", "field": "user_name", "align": "left"},
                        {"name": "borrow_date", "label": "借出日期", "field": "borrow_date", "align": "left"},
                        {"name": "due_date", "label": "应还日期", "field": "due_date", "align": "left"},
                        {"name": "status", "label": "状态", "field": "status_label", "align": "center"},
                        {"name": "action", "label": "操作", "field": "action", "align": "center"},
                    ]
                    rows = []
                    for r in records:
                        status_label = BORROW_STATUS.get(r["status"], (r["status"],))[0]
                        rows.append({
                            "id": r["id"],
                            "item_id": r["item_id"],
                            "item_code": r.get("item_code", ""),
                            "item_name": r.get("item_name", ""),
                            "user_name": r.get("user_name", ""),
                            "borrow_date": r.get("borrow_date", "")[:10],
                            "due_date": str(r.get("due_date", "")),
                            "status": r["status"],
                            "status_label": status_label,
                        })

                    table = ui.table(columns=columns, rows=rows, row_key="id").classes("w-full")

                    # 只有 active/overdue 状态且 (本人或管理员) 才能归还
                    table.add_slot("body-cell-action", '''
                        <q-td :props="props">
                            <q-btn v-if="props.row.status === 'active' || props.row.status === 'overdue'"
                                   flat dense color="green" label="归还"
                                   @click="$parent.$emit('return_item', props.row)" />
                            <q-btn flat dense color="blue" icon="visibility"
                                   @click="$parent.$emit('view_item', props.row)" />
                        </q-td>
                    ''')
                    table.on("return_item", lambda e: _show_return_dialog(e.args["id"], load_borrows))
                    table.on("view_item", lambda e: ui.navigate.to(f"/items/{e.args['item_id']}"))

                    # 分页
                    total_pages = (total + 19) // 20
                    if total_pages > 1:
                        with ui.row().classes("w-full justify-center q-mt-md"):
                            for p in range(1, total_pages + 1):
                                btn = ui.button(str(p), on_click=lambda _, pg=p: load_borrows(pg))
                                if p == page:
                                    btn.props("color=blue")
                                else:
                                    btn.props("flat")

            except Exception as e:
                with table_container:
                    ui.label(f"请求失败: {e}").classes("text-red")

        search_btn.on_click(lambda: load_borrows())
        load_borrows()


def _show_return_dialog(borrow_id: int, reload_fn):
    with ui.dialog() as dialog, ui.card().classes("w-96"):
        ui.label("归还物品").classes("text-h6")
        return_note = ui.input("物品状况备注", placeholder="如：完好无损").classes("full-width").props("outlined")
        msg = ui.label("").classes("text-red")

        with ui.row().classes("full-width justify-end gap-2"):
            ui.button("取消", on_click=dialog.close).props("flat")

            def do_return():
                result = api.return_item(borrow_id, return_note.value or "")
                if result.get("code") == 0:
                    dialog.close()
                    ui.notify("归还成功！", type="positive")
                    reload_fn()
                else:
                    msg.text = result.get("msg", "归还失败")

            ui.button("确认归还", on_click=do_return).props("color=green")
    dialog.open()

"""物品录入页（Admin）"""
import re
from nicegui import ui, app
from api_client import api
from components.navbar import navbar


def _normalize_date(s: str) -> str | None:
    """把各种日期格式统一为 YYYY-MM-DD"""
    s = s.strip()
    if not s:
        return None
    # 已经是 YYYY-MM-DD
    if re.match(r"^\d{4}-\d{2}-\d{2}$", s):
        return s
    # YYYYMMDD → YYYY-MM-DD
    if re.match(r"^\d{8}$", s):
        return f"{s[:4]}-{s[4:6]}-{s[6:8]}"
    # YYYY/MM/DD → YYYY-MM-DD
    if re.match(r"^\d{4}/\d{2}/\d{2}$", s):
        return s.replace("/", "-")
    return s  # 原样返回，让后端校验


@ui.page("/items/create")
def item_create_page():
    if not app.storage.user.get("user") or not app.storage.user.get("session_id"):
        ui.navigate.to("/login")
        return

    user = app.storage.user["user"]
    if user.get("role") != "admin":
        ui.navigate.to("/items")
        return

    navbar()

    with ui.column().classes("q-pa-md w-full max-w-3xl mx-auto"):
        with ui.row().classes("items-center gap-4"):
            ui.button(icon="arrow_back", on_click=lambda: ui.navigate.to("/items")).props("flat round")
            ui.label("录入物品").classes("text-h5")

        # 加载分类列表
        categories = {}
        try:
            result = api.get_categories()
            if result.get("code") == 0:
                for c in result["data"]:
                    categories[c["id"]] = c["name"]
        except Exception:
            pass

        with ui.card().classes("w-full q-mt-md q-pa-lg"):
            name = ui.input("物品名称 *", placeholder="如：投影仪").classes("full-width").props("outlined")
            category = ui.select(categories, label="分类").classes("full-width").props("outlined")
            model_input = ui.input("型号/规格", placeholder="如：EPSON CB-X06").classes("full-width").props("outlined")
            with ui.input("购入日期", placeholder="YYYY-MM-DD").classes("full-width").props("outlined") as purchase_date:
                with ui.menu().props("no-parent-event") as menu:
                    with ui.date(mask="YYYY-MM-DD").bind_value(purchase_date) as dp:
                        with ui.row().classes("justify-end"):
                            ui.button("确定", on_click=menu.close).props("flat")
                with purchase_date.add_slot("append"):
                    ui.icon("edit_calendar").on("click", menu.open).classes("cursor-pointer")
            purchase_price = ui.input("购入价格", placeholder="如：3500.00").classes("full-width").props("outlined type=number")
            location = ui.input("存放位置", placeholder="如：教学楼A栋 301室").classes("full-width").props("outlined")
            remark = ui.textarea("备注").classes("full-width").props("outlined")

            msg = ui.label("").classes("text-red q-mt-sm")

            def do_create():
                if not name.value:
                    msg.text = "请填写物品名称"
                    return

                data = {"name": name.value}
                if category.value:
                    data["category_id"] = category.value
                if model_input.value:
                    data["model"] = model_input.value
                if purchase_date.value:
                    d = _normalize_date(purchase_date.value)
                    if d:
                        data["purchase_date"] = d
                if purchase_price.value:
                    try:
                        data["purchase_price"] = float(purchase_price.value)
                    except ValueError:
                        msg.text = "购入价格格式不正确"
                        return
                if location.value:
                    data["location"] = location.value
                if remark.value:
                    data["remark"] = remark.value
                try:
                    result = api.create_item(data)
                    if result.get("code") == 0:
                        ui.notify("物品录入成功！", type="positive")
                        item_id = result["data"]["id"]
                        ui.navigate.to(f"/items/{item_id}")
                    else:
                        msg.text = result.get("msg", "录入失败")
                except Exception as e:
                    msg.text = f"请求失败: {e}"

            with ui.row().classes("full-width justify-end gap-2 q-mt-md"):
                ui.button("取消", on_click=lambda: ui.navigate.to("/items")).props("flat")
                ui.button("确认录入", icon="save", on_click=do_create).props("color=blue")

"""用户管理页（Admin only）"""
from nicegui import ui, app
from api_client import api
from components.navbar import navbar


ROLE_LABELS = {"admin": "管理员", "staff": "教职工", "student": "学生"}


@ui.page("/users")
def users_page():
    if not app.storage.user.get("user") or not app.storage.user.get("session_id"):
        ui.navigate.to("/login")
        return

    user = app.storage.user["user"]
    if user.get("role") != "admin":
        ui.navigate.to("/dashboard")
        return

    navbar()

    with ui.column().classes("q-pa-md w-full max-w-6xl mx-auto"):
        with ui.row().classes("w-full items-center justify-between"):
            ui.label("用户管理").classes("text-h5")
            ui.button("添加用户", icon="person_add", on_click=lambda: _show_create_dialog(load_users)).props("color=blue")

        table_container = ui.column().classes("w-full q-mt-md")

        def load_users(page=1):
            table_container.clear()

            try:
                result = api.list_users(page=page, page_size=20)
                if result.get("code") != 0:
                    with table_container:
                        ui.label(f"加载失败: {result.get('msg')}").classes("text-red")
                    return

                data = result["data"]
                users = data.get("items", [])
                total = data.get("total", 0)

                with table_container:
                    if not users:
                        ui.label("暂无用户").classes("text-grey text-center q-pa-lg")
                        return

                    columns = [
                        {"name": "username", "label": "用户名", "field": "username", "align": "left"},
                        {"name": "real_name", "label": "姓名", "field": "real_name", "align": "left"},
                        {"name": "role", "label": "角色", "field": "role_label", "align": "center"},
                        {"name": "phone", "label": "电话", "field": "phone", "align": "left"},
                        {"name": "is_active", "label": "状态", "field": "active_label", "align": "center"},
                        {"name": "action", "label": "操作", "field": "action", "align": "center"},
                    ]
                    rows = []
                    for u in users:
                        rows.append({
                            "id": u["id"],
                            "username": u["username"],
                            "real_name": u["real_name"],
                            "role": u["role"],
                            "role_label": ROLE_LABELS.get(u["role"], u["role"]),
                            "phone": u.get("phone") or "-",
                            "is_active": u.get("is_active", True),
                            "active_label": "启用" if u.get("is_active", True) else "已禁用",
                        })

                    table = ui.table(columns=columns, rows=rows, row_key="id").classes("w-full")
                    table.add_slot("body-cell-action", '''
                        <q-td :props="props">
                            <q-btn flat dense color="blue" label="编辑"
                                   @click="$parent.$emit('edit', props.row)" class="q-mr-xs" />
                            <q-btn v-if="props.row.is_active"
                                   flat dense color="red" label="禁用"
                                   @click="$parent.$emit('disable', props.row)" />
                        </q-td>
                    ''')
                    table.on("edit", lambda e: _show_edit_dialog(e.args, load_users))
                    table.on("disable", lambda e: _do_disable(e.args["id"], load_users))

                    # 分页
                    total_pages = (total + 19) // 20
                    if total_pages > 1:
                        with ui.row().classes("w-full justify-center q-mt-md"):
                            for p in range(1, total_pages + 1):
                                btn = ui.button(str(p), on_click=lambda _, pg=p: load_users(pg))
                                if p == page:
                                    btn.props("color=blue")
                                else:
                                    btn.props("flat")

            except Exception as e:
                with table_container:
                    ui.label(f"请求失败: {e}").classes("text-red")

        load_users()


def _show_create_dialog(reload_fn):
    with ui.dialog() as dialog, ui.card().classes("w-96"):
        ui.label("添加用户").classes("text-h6")
        username = ui.input("用户名 *").classes("full-width").props("outlined")
        password = ui.input("密码 *", password=True).classes("full-width").props("outlined")
        real_name = ui.input("真实姓名 *").classes("full-width").props("outlined")
        role = ui.select({"student": "学生", "staff": "教职工", "admin": "管理员"}, value="student", label="角色").classes("full-width").props("outlined")
        phone = ui.input("电话").classes("full-width").props("outlined")
        msg = ui.label("").classes("text-red")

        with ui.row().classes("full-width justify-end gap-2"):
            ui.button("取消", on_click=dialog.close).props("flat")

            def do_create():
                if not username.value or not password.value or not real_name.value:
                    msg.text = "请填写必填项"
                    return
                data = {
                    "username": username.value,
                    "password": password.value,
                    "real_name": real_name.value,
                    "role": role.value,
                }
                if phone.value:
                    data["phone"] = phone.value

                result = api.create_user(data)
                if result.get("code") == 0:
                    dialog.close()
                    ui.notify("用户创建成功", type="positive")
                    reload_fn()
                else:
                    msg.text = result.get("msg", "创建失败")

            ui.button("创建", on_click=do_create).props("color=blue")
    dialog.open()


def _show_edit_dialog(user_data: dict, reload_fn):
    user_id = user_data["id"]
    with ui.dialog() as dialog, ui.card().classes("w-96"):
        ui.label("编辑用户").classes("text-h6")
        ui.label(f"用户名: {user_data['username']}").classes("text-grey")
        real_name = ui.input("真实姓名", value=user_data["real_name"]).classes("full-width").props("outlined")
        role = ui.select(
            {"student": "学生", "staff": "教职工", "admin": "管理员"},
            value=user_data["role"],
            label="角色",
        ).classes("full-width").props("outlined")
        phone = ui.input("电话", value=user_data.get("phone") or "").classes("full-width").props("outlined")
        msg = ui.label("").classes("text-red")

        with ui.row().classes("full-width justify-end gap-2"):
            ui.button("取消", on_click=dialog.close).props("flat")

            def do_update():
                data = {}
                if real_name.value:
                    data["real_name"] = real_name.value
                if role.value:
                    data["role"] = role.value
                if phone.value:
                    data["phone"] = phone.value

                result = api.update_user(user_id, data)
                if result.get("code") == 0:
                    dialog.close()
                    ui.notify("用户更新成功", type="positive")
                    reload_fn()
                else:
                    msg.text = result.get("msg", "更新失败")

            ui.button("保存", on_click=do_update).props("color=blue")
    dialog.open()


def _do_disable(user_id: int, reload_fn):
    result = api.disable_user(user_id)
    if result.get("code") == 0:
        ui.notify("用户已禁用", type="warning")
        reload_fn()
    else:
        ui.notify(result.get("msg", "操作失败"), type="negative")
